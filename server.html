"""Serve the local Sprint 06 API documentation and simulated assistant endpoints.

The sandbox inherits Python's threaded HTTP server and serves the ``public``
directory for GET requests. POST requests use a fixed local bearer token and
operate on in-memory fixtures. Calendar actions follow proposal, confirmation,
and execution states; execution records a simulated provider write. No Google
calls, real OAuth, AI inference, external webhook delivery, or persistence occur.
Restarting the process clears all action and history state.

Examples:
    Validate the session identifier used throughout the documentation:

    >>> fields({"session_id": "session_anu_01"}, ["session_id"])
    >>> text_fields({"session_id": "session_anu_01"}, ["session_id"])
"""

from datetime import datetime, timezone
from email.parser import BytesParser
from email.policy import default

import json

TOKEN = "sprint05-local-sandbox"
#: int: Maximum declared request body size in bytes (2 MiB).
LIMIT = 2 * 1024 * 1024

class SandboxServer(ThreadingHTTPServer):
    """Host the HTTP sandbox and its shared, in-memory action state.

    Construction binds and activates a listening socket through
    ``http.server.ThreadingHTTPServer``; it does not start the serving
    loop. Call ``serve_forever()`` to accept requests and ``server_close()`` to
    release the socket. POST handlers serialize their work with ``lock``.

    Args:
        address (tuple[str, int]): Host and port to bind. Port ``0`` requests an
            available port from the operating system.

    Attributes:
        actions (dict[str, dict[str, object]]): Proposed and completed actions,
            keyed by generated action ID.
        history (dict[str, list[dict[str, object]]]): Executed action records,
            grouped by session ID.
        sequence (int): Counter used to generate POST correlation IDs.
        provider_calls (int): Number of simulated provider writes. Repeating a
            successful execution does not increment this count.
        lock (threading.Lock): Lock protecting the complete POST operation.

    Returns:
        SandboxServer: Bound server with empty action and history collections.

    Raises:
        OSError: If the listening socket cannot be created, bound, or activated.
        OverflowError: If the supplied port is outside the valid socket range.
        TypeError: If the address has an invalid type or structure.

    Examples:
        Bind an available local port, inspect initial state, and close without
        starting the serving loop or contacting an external service:

        >>> server = SandboxServer(("127.0.0.1", 0))
        >>> try:
        ...     print(server.actions, server.history, server.provider_calls)
        ... finally:
        ...     server.server_close()
        {} {} 0
    """

    def __init__(self, address):
        """Bind the inherited HTTP server and initialize shared sandbox state.

        Args:
            address (tuple[str, int]): Listening host and port, forwarded to
                ``ThreadingHTTPServer`` with ``Handler`` as the request class.

        Returns:
            None: Initializes the socket, state dictionaries, counters, and lock.

        Raises:
            OSError: If creating, binding, or activating the socket fails.
            OverflowError: If the port is outside the valid socket range.
            TypeError: If the address has an invalid type or structure.

        Examples:
            >>> server = SandboxServer(("127.0.0.1", 0))
            >>> try:
            ...     print(server.sequence, server.provider_calls)
            ... finally:
            ...     server.server_close()
            0 0
        """
        super().__init__(address, Handler)
        self.actions = {}
        self.history = {}
        self.sequence = 0
        self.provider_calls = 0
        self.lock = threading.Lock()

class RequestError(Exception):
    """Carry an HTTP status and a user-facing sandbox validation message.

    ``Handler.do_POST`` catches this exception and serializes its attributes
    into the documented error response. The constructor stores its arguments
    without validating their types or formatting the message.

    Args:
        status (int): HTTP error status, such as ``400`` or ``409``.
        message (str): Explanation suitable for the JSON ``user_message`` field.

    Attributes:
        status (int): HTTP status to send.
        message (str): User-facing explanation of the failure.

    Returns:
        RequestError: Exception carrying the supplied status and message.

    Notes:
        No exceptions are raised. Construction performs only attribute assignment.

    Examples:
        >>> error = RequestError(409, "Confirm Sprint 06 review before execution.")
        >>> print(error.status, error.message)
        409 Confirm Sprint 06 review before execution.
    """

    def __init__(self, status, message):
        """Store the HTTP status and message for the POST error handler.

        Args:
            status (int): HTTP error status to report.
            message (str): User-facing error explanation.

        Returns:
            None: Assigns ``status`` and ``message`` to the exception instance.

        Notes:
            No exceptions are raised. No application-level validation is performed.

        Examples:
            >>> error = RequestError(400, "session_id must be a nonempty string.")
            >>> error.status, error.message
            (400, 'session_id must be a nonempty string.')
        """
        self.status, self.message = status, message

def fields(payload, names):
    """Require a dictionary with exactly the specified field names.

    Values are not inspected. Call ``text_fields`` separately to validate
    string content where needed.

    Args:
        payload (dict[str, object]): Request object whose keys must match.
        names (list[str] | tuple[str, ...]): Required field names. Key order
            does not affect validation; this order appears in error messages.

    Returns:
        None: Successful validation leaves the payload unchanged.

    Raises:
        RequestError: With status ``400`` if the payload is not a dictionary or
            has missing or extra fields.
        TypeError: If ``names`` is not iterable, contains unhashable elements,
            or contains non-string elements needed for an error message.

    Examples:
        >>> payload = {"session_id": "session_anu_01"}
        >>> fields(payload, ["session_id"])
        >>> payload
        {'session_id': 'session_anu_01'}
    """
    if not isinstance(payload, dict) or set(payload) != set(names):
        raise RequestError(400, "Provide exactly these fields: " + ", ".join(names))

def text_fields(payload, names):
    """Require each selected field to contain a nonempty string.

    Whitespace is used only to detect empty content; original values are not
    trimmed or otherwise changed. This helper assumes the fields already
    exist, normally after a successful call to ``fields``.

    Args:
        payload (dict[str, object]): Request object containing the named fields.
        names (list[str] | tuple[str, ...]): Keys whose values must be strings
            with at least one non-whitespace character.

    Returns:
        None: Successful validation leaves all field values unchanged.

    Raises:
        RequestError: With status ``400`` if a selected value is not a string
            or contains only whitespace.
        KeyError: If a selected field is missing from the payload.
        TypeError: If the supplied payload or field-name collection cannot
            support the expected iteration and dictionary lookup operations.

    Examples:
        >>> payload = {"session_id": "session_anu_01", "title": "Sprint 06 review"}
        >>> fields(payload, ["session_id", "title"])
        >>> text_fields(payload, ["session_id", "title"])
        >>> payload["title"]
        'Sprint 06 review'
    """
    for name in names:
        if not isinstance(payload[name], str) or not payload[name].strip():
            raise RequestError(400, name + " must be a nonempty string.")

class Handler(SimpleHTTPRequestHandler):
    """Serve static documentation and implement the sandbox POST API.

    Inherited GET/HEAD behavior serves ``ROOT / "public"``. The server creates
    one handler for an accepted client connection; inherited initialization
    processes that connection immediately. Helpers can be demonstrated without
    network activity by creating an inert instance with ``__new__`` and setting
    the state each helper requires.

    Args:
        *args (object): Positional arguments forwarded to the inherited request
            constructor: client socket, client address, and server instance.
        **kwargs (object): Supported inherited constructor keyword arguments,
            such as ``request``, ``client_address``, and ``server`` when not
            supplied positionally. Do not supply ``directory``; this class
            supplies its static documentation directory explicitly.

    Attributes:
        server (SandboxServer): Server providing shared action state and lock.
        correlation (str): Generated correlation ID, assigned by ``do_POST``.
        request_bytes (bytes): Capped request body read by ``do_POST`` before
            ``parse_body`` is called.

    Returns:
        Handler: Handler that has processed its supplied client connection.

    Raises:
        TypeError: If inherited constructor arguments are missing, duplicated,
            unsupported, or include a second ``directory`` value.
        OSError: If inherited socket setup, request processing, or cleanup fails.

    Examples:
        >>> handler = Handler.__new__(Handler)
        >>> handler.correlation = "corr_sandbox_0001"
        >>> handler.envelope("proposed", "Review Sprint 06 review before confirming.")
        {'status': 'proposed', 'user_message': 'Review Sprint 06 review before confirming.', 'correlation_id': 'corr_sandbox_0001'}
    """

    def __init__(self, *args, **kwargs):
        """Process a connection using the fixed static documentation directory.

        This forwards to ``SimpleHTTPRequestHandler`` and ultimately
        ``BaseRequestHandler(request, client_address, server)``. It handles the
        connection during construction rather than deferring processing.

        Args:
            *args (object): Inherited positional arguments: client socket,
                client address, and server, in that order.
            **kwargs (object): Equivalent inherited keyword arguments when not
                already positional. ``directory`` is supplied by this method
                and must not also appear here.

        Returns:
            None: Initializes and processes the inherited request handler.

        Raises:
            TypeError: If forwarded arguments are invalid, missing, duplicated,
                or include a conflicting ``directory`` argument.
            OSError: If inherited socket setup, request handling, or cleanup
                cannot complete.

        Examples:
            Mock inherited initialization to inspect the directory argument
            without opening or processing a connection:

            >>> from unittest.mock import patch
            >>> with patch.object(SimpleHTTPRequestHandler, "__init__", return_value=None) as initialize:
            ...     handler = Handler("client-socket", ("127.0.0.1", 12345), "sandbox-server")
            ...     print(initialize.call_args.kwargs["directory"] == str(ROOT / "public"))
            True
        """
        super().__init__(*args, directory=str(ROOT / "public"), **kwargs)

    def send_json(self, code, body):
        """Write a UTF-8 JSON HTTP response with caching disabled.

        Args:
            code (int): HTTP status code passed to the inherited response writer.
            body (object): JSON-serializable response value. Non-ASCII text is
                emitted directly in UTF-8.

        Returns:
            None: Writes status, content headers, and encoded body to ``wfile``.

        Raises:
            TypeError: If the body is not JSON-serializable or the status code
                is incompatible with inherited HTTP formatting.
            ValueError: If JSON serialization finds a circular container.
            UnicodeError: If text cannot be encoded as UTF-8.
            OSError: If the response stream cannot be written.
            AttributeError: If inherited request state or the output stream
                has not been initialized.

        Examples:
            Use an in-memory stream with the inherited response writer:

            >>> from io import BytesIO
            >>> handler = Handler.__new__(Handler)
            >>> handler.request_version, handler.command = "HTTP/1.1", "POST"
            >>> handler.requestline = "POST /v1/sessions/history HTTP/1.1"
            >>> handler.wfile = BytesIO()
            >>> handler.send_json(200, {"session_id": "session_anu_01"})
            >>> json.loads(handler.wfile.getvalue().splitlines()[-1])
            {'session_id': 'session_anu_01'}
        """
        encoded = json.dumps(body, ensure_ascii=False).encode("utf-8")
        self.send_response(code)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(encoded)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(encoded)

    def envelope(self, status, message):
        """Create the common response fields for the current POST request.

        Args:
            status (str): Result state, such as ``proposed``, ``confirmed``,
                ``succeeded``, or ``error``.
            message (str): User-facing description of the result.

        Returns:
            dict[str, str]: ``status``, ``user_message``, and the handler's
            current ``correlation_id``. Values are copied without validation.

        Raises:
            AttributeError: If ``correlation`` has not yet been assigned.

        Examples:
            >>> handler = Handler.__new__(Handler)
            >>> handler.correlation = "corr_sandbox_0001"
            >>> result = handler.envelope("proposed", "Review Sprint 06 review.")
            >>> result["status"], result["correlation_id"]
            ('proposed', 'corr_sandbox_0001')
        """
        return {"status": status, "user_message": message, "correlation_id": self.correlation}

    def parse_body(self):
        """Validate the declared body size and decode the previously read bytes.

        ``do_POST`` must first populate ``request_bytes``. This helper does not
        read from ``rfile`` or verify that the actual byte count equals the
        declared length. It accepts JSON for ordinary paths and multipart form
        data for the photo and bark-audio upload paths, and sets a ten-second
        timeout on the connection.

        Returns:
            object: Decoded JSON value of any JSON type, or a multipart field
            dictionary. Multipart text values are UTF-8 strings; file values
            are dictionaries containing basename ``filename``, MIME
            ``content_type``, and raw ``data`` bytes. Endpoint-specific object
            and field validation occurs later in ``dispatch``.

        Raises:
            RequestError: With ``400`` for absent/invalid declared lengths,
                malformed JSON or multipart structure, or duplicate multipart
                fields; ``413`` for a declared length over ``LIMIT``; or
                ``415`` for an unsupported content type.
            UnicodeError: If a multipart text field cannot be decoded as UTF-8.
            OSError: If setting the connection timeout fails.
            AttributeError: If required handler state such as ``request_bytes``
                has not been initialized.

        Examples:
            Decode a history request using an inert handler and mocked socket:

            >>> from unittest.mock import Mock
            >>> handler = Handler.__new__(Handler)
            >>> handler.path = "/v1/sessions/history"
            >>> handler.request_bytes = b'{"session_id": "session_anu_01"}'
            >>> handler.headers = {"Content-Length": str(len(handler.request_bytes)), "Content-Type": "application/json"}
            >>> handler.connection = Mock()
            >>> handler.parse_body()
            {'session_id': 'session_anu_01'}
        """
        try:
            size = int(self.headers.get("Content-Length", "0"))
        except ValueError:
            raise RequestError(400, "Invalid Content-Length.")
        if size <= 0:
            raise RequestError(400, "A request body is required.")
        if size > LIMIT:
            raise RequestError(413, "Request exceeds 2 MiB.")
        self.connection.settimeout(10)
        raw = self.request_bytes
        content_type = self.headers.get("Content-Type", "")
        if self.path in ("/v1/pets/upload-photo", "/v1/audio/translate-bark"):
            if not content_type.startswith("multipart/form-data;"):
                raise RequestError(415, "Use multipart/form-data with a boundary.")
            message = BytesParser(policy=default).parsebytes(
                ("Content-Type: " + content_type + "\r\nMIME-Version: 1.0\r\n\r\n").encode() + raw)
            if not message.is_multipart():
                raise RequestError(400, "Malformed multipart body.")
            payload = {}
            for part in message.iter_parts():
                name = part.get_param("name", header="Content-Disposition")
                if name in payload:
                    raise RequestError(400, "Duplicate multipart field.")
                data = part.get_payload(decode=True)
                if part.get_filename():
                    payload[name] = {"filename": Path(part.get_filename()).name,
                                     "content_type": part.get_content_type(), "data": data}
                else:
                    payload[name] = data.decode("utf-8")
            return payload
        if content_type.split(";")[0] != "application/json":
            raise RequestError(415, "Use application/json.")
        try:
            return json.loads(raw)
        except (ValueError, UnicodeError):
            raise RequestError(400, "Malformed JSON body.")

    def do_POST(self):
        """Authenticate, parse, dispatch, and respond to one sandbox POST request.

        The server lock protects correlation numbering and endpoint state for
        the complete operation. Request reads are capped at ``LIMIT + 1`` bytes.
        Authorization must be ``Bearer `` followed by ``TOKEN``; a missing or
        invalid value produces ``401``. The optional ``X-Sandbox-Failure`` header
        simulates ``403``, ``500``, or ``503`` before dispatch. ``RequestError``
        becomes its specified HTTP error response;
        field/JSON type errors become ``400`` and other processing exceptions
        become ``500``.

        Returns:
            None: Writes one JSON response when the connection remains usable.

        Raises:
            OSError: If writing an error response also fails, for example after
                a client disconnect. Errors while sending the normal response
                first trigger an attempted ``500`` response.
            AttributeError: If the handler lacks server lock/counter state
                needed before entering its exception-handling block.

        Examples:
            Capture a response using in-memory request data and test doubles:

            >>> from io import BytesIO
            >>> from types import SimpleNamespace
            >>> from unittest.mock import Mock
            >>> handler = Handler.__new__(Handler)
            >>> handler.server = SimpleNamespace(lock=threading.Lock(), sequence=0, history={})
            >>> handler.path = "/v1/sessions/history"
            >>> raw = b'{"session_id": "session_anu_01"}'
            >>> handler.headers = {"Content-Length": str(len(raw)), "Content-Type": "application/json", "Authorization": "Bearer " + TOKEN}
            >>> handler.rfile, handler.connection = BytesIO(raw), Mock()
            >>> handler.send_json = Mock()
            >>> handler.do_POST()
            >>> code, response = handler.send_json.call_args.args
            >>> code, response["status"], response["items"]
            (200, 'succeeded', [])
        """
        with self.server.lock:
            self.server.sequence += 1
            self.correlation = f"corr_sandbox_{self.server.sequence:04d}"
            try:
                # Consume ordinary request bodies before early error responses on Windows.
                # Cap the read so oversized uploads never allocate unbounded memory.
                self.connection.settimeout(10)
                size = int(self.headers.get("Content-Length", "0"))
                if size < 0:
                    raise RequestError(400, "Invalid Content-Length.")
                self.request_bytes = self.rfile.read(min(size, LIMIT + 1))
                if self.headers.get("Authorization") != "Bearer " + TOKEN:
                    raise RequestError(401, "Missing or invalid sandbox bearer token.")
                failure = self.headers.get("X-Sandbox-Failure")
                if failure in ("403", "500", "503"):
                    raise RequestError(int(failure), {"403": "Sandbox provider permission is unavailable.",
                        "500": "Unexpected internal sandbox error.", "503": "Sandbox dependency unavailable."}[failure])
                payload = self.parse_body()
                code, result = self.dispatch(payload)
                self.send_json(code, result)
            except RequestError as error:
                self.send_json(error.status, {**self.envelope("error", error.message),
                    "error": {"code": "http_" + str(error.status), "message": error.message}})
            except (UnicodeError, ValueError, KeyError, TypeError):
                self.send_json(400, {**self.envelope("error", "Review the request field types and values."),
                    "error": {"code": "http_400", "message": "Review the request field types and values."}})
            except Exception:
                self.send_json(500, {**self.envelope("error", "Unexpected internal sandbox error."),
                    "error": {"code": "http_500", "message": "Unexpected internal sandbox error."}})

    def dispatch(self, payload):
        """Execute the endpoint selected by ``path`` against sandbox fixtures.

        Calendar requests propose actions; confirmation requires the matching
        session and payload version. Execution requires confirmation and is
        idempotent once successful: retries preserve one history record and one
        simulated provider write. Other routes search fixture contacts, read
        session history, inspect fixture uploads, or return a simulated webhook
        subscription. No route invokes an external provider.

        Authentication and locking belong to ``do_POST``; callers using this
        helper directly must supply initialized server state and coordinate any
        concurrent access themselves.

        Args:
            payload (dict[str, object]): Parsed endpoint request. Commands need
                exactly ``session_id``, ``transcript``, ``intent``, and ``event``;
                the supported intent is ``calendar.create``. Its event needs
                ``title``, timezone-aware ISO ``start``/``end`` with end after
                start, and ``timezone`` equal to ``Asia/Ulaanbaatar``. Action
                confirmation/execution need ``session_id`` and positive integer
                ``payload_version``. Other routes enforce their own exact field
                sets. Uploads use the file dictionaries from ``parse_body``.

        Returns:
            tuple[int, dict[str, object]]: HTTP status and JSON response object.
            Successful responses use ``200``, except webhook subscription
            creation uses ``201``. Command/action, contact, and history results
            include the common envelope; upload and subscription results have
            their endpoint-specific fields.

        Raises:
            RequestError: With ``400`` for request validation failures, ``404``
                for unknown endpoints/resources/session actions, ``409`` for
                action version/state conflicts, or ``415`` for invalid upload
                fixture types/signatures.
            ValueError: If a datetime or callback URL cannot be parsed.
            KeyError: If an upload dictionary lacks an accessed metadata field.
            TypeError: If nested upload data has an incompatible value type.
            AttributeError: If upload data lacks the expected byte operations,
                or required handler/server state has not been initialized.

        Examples:
            Propose, confirm, and execute an action entirely in memory. Repeating
            execution does not duplicate the simulated provider write:

            >>> from types import SimpleNamespace
            >>> handler = Handler.__new__(Handler)
            >>> handler.server = SimpleNamespace(actions={}, history={}, provider_calls=0)
            >>> handler.correlation = "corr_sandbox_0001"
            >>> handler.path = "/v1/commands"
            >>> payload = {
            ...     "session_id": "session_anu_01",
            ...     "transcript": "Create Sprint 06 review at 10:00 in Ulaanbaatar.",
            ...     "intent": "calendar.create",
            ...     "event": {"title": "Sprint 06 review", "start": "2026-10-12T10:00+08:00",
            ...               "end": "2026-10-12T10:30+08:00", "timezone": "Asia/Ulaanbaatar"},
            ... }
            >>> code, proposal = handler.dispatch(payload)
            >>> code, proposal["action_id"], proposal["status"]
            (200, 'action_0001', 'proposed')
            >>> action_request = {"session_id": "session_anu_01", "payload_version": 1}
            >>> handler.path = "/v1/actions/action_0001/confirm"
            >>> code, result = handler.dispatch(action_request)
            >>> result["status"]
            'confirmed'
            >>> handler.path = "/v1/actions/action_0001/execute"
            >>> code, result = handler.dispatch(action_request)
            >>> code, result = handler.dispatch(action_request)
            >>> result["status"], handler.server.provider_calls, len(handler.server.history["session_anu_01"])
            ('succeeded', 1, 1)
        """
        path = self.path
        if path == "/v1/commands":
            fields(payload, ["session_id", "transcript", "intent", "event"])
            text_fields(payload, ["session_id", "transcript", "intent"])
            if payload["intent"] != "calendar.create":
                raise RequestError(400, "This sandbox supports calendar.create only.")
            event = payload["event"]
            fields(event, ["title", "start", "end", "timezone"])
            text_fields(event, ["title", "start", "end", "timezone"])
            start, end = [datetime.fromisoformat(event[key].replace("Z", "+00:00")) for key in ["start", "end"]]
            if start.tzinfo is None or end.tzinfo is None or end <= start or event["timezone"] != "Asia/Ulaanbaatar":
                raise RequestError(400, "Use timezone-aware start/end, end after start, and Asia/Ulaanbaatar.")
            action_id = f"action_{len(self.server.actions)+1:04d}"
            self.server.actions[action_id] = {**payload, "payload_version": 1, "status": "proposed"}
            return 200, {**self.envelope("proposed", "Review this action before confirming."),
                "transcript": payload["transcript"], "intent": payload["intent"], "action_id": action_id,
                "payload_version": 1, "required_fields": [], "event": event}
        match = re.fullmatch(r"/v1/actions/([a-zA-Z0-9_]+)/(?P<verb>confirm|execute)", path)
        if match:
            fields(payload, ["session_id", "payload_version"])
            text_fields(payload, ["session_id"])
            if type(payload["payload_version"]) is not int or payload["payload_version"] < 1:
                raise RequestError(400, "payload_version must be a positive integer.")
            action_id, verb = match.group(1), match.group("verb")
            action = self.server.actions.get(action_id)
            if not action or action["session_id"] != payload["session_id"]:
                raise RequestError(404, "Resource or session action not found.")
            if action["payload_version"] != payload["payload_version"]:
                raise RequestError(409, "Action version or confirmation conflicts.")
            if verb == "confirm":
                if action["status"] not in ("proposed", "confirmed"):
                    raise RequestError(409, "Completed actions cannot be reconfirmed.")
                action["status"] = "confirmed"
            else:
                if action["status"] not in ("confirmed", "succeeded"):
                    raise RequestError(409, "Confirm the exact action before execution.")
                if action["status"] == "confirmed":
                    self.server.provider_calls += 1  # Simulated provider write count, no network.
                    action["status"] = "succeeded"
                    self.server.history.setdefault(action["session_id"], []).append({"action_id": action_id,
                        "type": action["intent"], "status": "succeeded",
                        "occurred_at": datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")})
            return 200, {**self.envelope(action["status"], "Sandbox action " + action["status"] + "."),
                "action_id": action_id, "payload_version": 1,
                "provider_resource_id": "sandbox_event_" + action_id if action["status"] == "succeeded" else ""}
        if path == "/v1/contacts/search":
            fields(payload, ["query"]); text_fields(payload, ["query"])
            contacts = [{"name": "Anu", "email": "anu@example.org"}] if "anu" in payload["query"].lower() else []
